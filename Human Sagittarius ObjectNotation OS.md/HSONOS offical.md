#!/usr/bin/env python3
import json
import re
import sys
from pathlib import Path

HERE = Path(__file__).resolve().parent
DEFAULT_SRC = HERE / "mathos.txt"
DEFAULT_OUT = HERE / "compiled"

MARKER = re.compile(r"^hson(lisp)?$")
HEADER = re.compile(r"^(HSON(?:_[A-Z]+)?)\s+([A-Za-z0-9_]+)\s*\{")
TOKEN = re.compile(
    r'"(?:[^"\\]|\\.)*"'
    r"|->"
    r"|>=|<=|==|!="
    r"|[{}()\[\],:=]"
    r"|[^\s{}()\[\],:=]+"
)
CLOSER = {"}", "]", ",", ")"}
NUM_INT = re.compile(r"^-?\d+$")
NUM_FLOAT = re.compile(r"^-?\d+\.\d+$")


def strip_comment(line):
    out = []
    in_str = False
    i = 0
    while i < len(line):
        c = line[i]
        if c == '"':
            in_str = not in_str
            out.append(c)
        elif not in_str and line[i : i + 2] == "//":
            break
        elif not in_str and c == ";":
            break
        else:
            out.append(c)
        i += 1
    return "".join(out)


def brace_delta(stripped):
    delta = 0
    in_str = False
    for c in stripped:
        if c == '"':
            in_str = not in_str
        elif not in_str and c == "{":
            delta += 1
        elif not in_str and c == "}":
            delta -= 1
    return delta


def extract_blocks(text):
    lines = text.splitlines()
    blocks = []
    warnings = []
    i = 0
    while i < len(lines):
        if MARKER.match(lines[i].strip()):
            j = i + 1
            while j < len(lines) and not lines[j].strip():
                j += 1
            header_line = lines[j].strip() if j < len(lines) else ""
            m = HEADER.match(header_line)
            if not m:
                i += 1
                continue
            lang = "hsonlisp" if MARKER.match(lines[i].strip()).group(1) else "hson"
            depth = 0
            started = False
            buf = []
            while j < len(lines):
                stripped = strip_comment(lines[j])
                buf.append(lines[j])
                d = brace_delta(stripped)
                depth += d
                if d > 0:
                    started = True
                if started and depth <= 0:
                    break
                j += 1
            if not started:
                warnings.append(f"unterminated block near line {i + 1}")
            blocks.append({"lang": lang, "header_kind": m.group(1),
                           "name": m.group(2), "source": "\n".join(buf)})
            i = j + 1
        else:
            i += 1
    return blocks, warnings


def tokenize(src):
    clean = "\n".join(strip_comment(l) for l in src.splitlines())
    return TOKEN.findall(clean)


class Parser:
    def __init__(self, toks):
        self.t = toks
        self.i = 0

    def peek(self, k=0):
        p = self.i + k
        return self.t[p] if p < len(self.t) else None

    def take(self):
        tok = self.peek()
        self.i += 1
        return tok

    def raw_form(self):
        assert self.take() == "("
        depth = 1
        parts = ["("]
        while depth > 0:
            tok = self.take()
            if tok is None:
                break
            if tok == "(":
                depth += 1
            elif tok == ")":
                depth -= 1
            parts.append(tok)
        return " ".join(parts)

    def skip_to_close(self, open_tok="{", close_tok="}"):
        start = self.i
        depth = 0
        while self.peek() is not None:
            tok = self.peek()
            if tok == open_tok:
                depth += 1
            elif tok == close_tok:
                depth -= 1
                if depth == 0:
                    break
            self.i += 1
        return self.t[start : self.i]

    def parse_body(self):
        assert self.take() == "{"
        items = []
        while True:
            tok = self.peek()
            if tok is None:
                break
            if tok == "}":
                self.take()
                break
            before = self.i
            item = self.parse_item()
            if self.i == before:
                self.take()
                items.append({"atom": "?stuck"})
            elif item is not None:
                items.append(item)
        return items

    def parse_item(self):
        head = []
        while True:
            tok = self.peek()
            if tok is None or tok in CLOSER or tok in "{[(:=":
                break
            head.append(self.take())
        term = self.peek()
        if term is None:
            return {"atom": " ".join(head)} if head else {"atom": ""}
        if term == "{":
            body = self.parse_body()
            if len(head) == 1:
                return {"block": head[0], "body": body}
            if not head:
                return {"block": "_anon", "body": body}
            return {"kind": head[0], "name": "_".join(head[1:]), "body": body}
        if term == "(":
            form = self.raw_form()
            nxt = self.peek()
            body = self.parse_body() if nxt == "{" else []
            if not head:
                return {"form": form, "body": body} if body else {"form": form}
            return {"kind": head[0], "name": "_".join(head[1:]),
                    "params": form, "body": body}
        if term == "=":
            self.take()
            key = "_".join(head) if head else "?"
            return {"key": key, "value": self.parse_value()}
        if term == ":":
            self.take()
            tparts = []
            while True:
                tok = self.peek()
                if tok is None or tok in CLOSER or tok in "{[(:=":
                    break
                if self.peek(1) == ":":
                    break
                tparts.append(self.take())
            ftype = " ".join(tparts)
            body = self.parse_body() if self.peek() == "{" else None
            item = {"field": head[0] if head else "?", "type": ftype}
            if body is not None:
                item["body"] = body
            return item
        if term == ",":
            self.take()
            return None if not head else {"atom": " ".join(head)}
        if term == "[":
            return {"atom": " ".join(head), "list": self.parse_list()}
        return {"atom": " ".join(head)}

    def parse_list(self):
        assert self.take() == "["
        vals = []
        while True:
            tok = self.peek()
            if tok is None:
                break
            if tok == "]":
                self.take()
                break
            if tok == ",":
                self.take()
                continue
            vals.append(self.parse_value())
        return vals

    def parse_map_literal(self):
        assert self.take() == "{"
        mapping = {}
        order = []
        while True:
            tok = self.peek()
            if tok is None:
                break
            if tok == "}":
                self.take()
                break
            if tok == ",":
                self.take()
                continue
            key = self.take()
            if self.peek() == "->" or self.peek() in ("→", "↔"):
                self.take()
                vparts = []
                while True:
                    vt = self.peek()
                    if vt is None or vt in ("}", ",") or vt in ("→", "↔"):
                        break
                    vparts.append(self.take())
                val = " ".join(vparts)
                if key not in mapping:
                    order.append(key)
                mapping[key] = val
            else:
                vparts = [key]
                while self.peek() not in (None, "}", ","):
                    vparts.append(self.take())
                key2 = " ".join(vparts)
                if key2 not in mapping:
                    order.append(key2)
                mapping[key2] = ""
        return {"_map": mapping, "_order": order}

    def has_top_arrow(self, span):
        depth = 0
        prev = None
        for tok in span:
            if tok in "{[(":
                depth += 1
            elif tok in "}])":
                depth -= 1
            elif depth == 0 and tok in ("→", "↔") and prev != "=":
                return True
            prev = tok
        return False

    def parse_value(self):
        tok = self.peek()
        if tok is None:
            return ""
        if tok.startswith('"'):
            self.take()
            return tok[1:-1]
        if tok == "(":
            return self.raw_form()
        if tok == "[":
            return self.parse_list()
        if tok == "{":
            span = self.skip_to_close()
            if self.has_top_arrow(span):
                return self.parse_map_literal()
            self.take()
            inner = Parser(span[1:-1])
            items = []
            while inner.peek() is not None:
                if inner.peek() == ",":
                    inner.take()
                    continue
                items.append(inner.parse_item())
            return {"_items": items}
        parts = []
        while True:
            tok = self.peek()
            if tok is None or tok in CLOSER or tok in "{[(:=":
                break
            nk = self.peek(1)
            if nk in ("=", ":", "{") and tok not in ("->", "→", "↔"):
                break
            parts.append(self.take())
        if not parts:
            return self.take() or ""
        text = " ".join(parts)
        if NUM_INT.match(text):
            return int(text)
        if NUM_FLOAT.match(text):
            return float(text)
        return text


def body_to_obj(items):
    obj = {}
    for item in items:
        if "key" in item:
            k, v = item["key"], item["value"]
        elif "field" in item:
            k, v = item["field"], item.get("type", "")
            if "body" in item:
                v = {"type": item["type"], "body": body_to_obj(item["body"])}
        elif "block" in item:
            k, v = item["block"], body_to_obj(item["body"])
        elif "kind" in item:
            k = item["kind"]
            v = {"name": item.get("name", ""), "body": body_to_obj(item["body"])}
            if "params" in item:
                v["params"] = item["params"]
        elif "form" in item:
            k, v = "_form", item["form"]
        elif "list" in item:
            k, v = "_anon", {"atom": item["atom"], "list": item["list"]}
        else:
            k, v = "_anon", item.get("atom", "")
        if k in obj:
            if not isinstance(obj[k], list):
                obj[k] = [obj[k]]
            obj[k].append(v)
        else:
            obj[k] = v
    return obj


def compile_block(block, index):
    toks = tokenize(block["source"])
    p = Parser(toks)
    head = []
    while p.peek() is not None and p.peek() != "{":
        head.append(p.take())
    items = p.parse_body()
    ast = body_to_obj(items)
    leftover = p.peek()
    doc = {
        "id": index,
        "language": block["lang"],
        "header_kind": block["header_kind"],
        "unit": block["name"],
        "ast": ast,
    }
    ok = leftover is None and bool(ast)
    return doc, ok, ("tokens left: %s" % leftover) if leftover else ""
