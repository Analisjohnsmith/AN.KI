                    next[idx] += self.strength * (gx + gy) * dt;
                }
            }
            mem.data = next;
        }
        }

        #[derive(Clone)]
        pub struct Field {
        pub data: Vec<f32>,
        pub width: usize,
        pub height: usize,
        }
        pub fn gradient(field: &Field, x: usize, y: usize) -> (f32, f32) {
        let w = field.width;
        let h = field.height;
        let idx = |x, y| y * w + x;
        let left = field.data[idx(x.saturating_sub(1), y)];
        let right = field.data[idx((x + 1).min(w - 1), y)];
        let up = field.data[idx(x, y.saturating_sub(1))];
        let down = field.data[idx(x, (y + 1).min(h - 1))];
        (right - left, down - up)
        }
        pub fn diffuse(field: &mut Field, rate: f32) {
        let mut next = field.data.clone();
        let w = field.width;
        let h = field.height;
        for y in 1..h - 1 {
            for x in 1..w - 1 {
                let idx = |x, y| y * w + x;
                let c = field.data[idx(x, y)];
                let u = field.data[idx(x, y - 1)];
                let d = field.data[idx(x, y + 1)];
                let l = field.data[idx(x - 1, y)];
                let r = field.data[idx(x + 1, y)];
                let lap = u + d + l + r - 4.0 * c;
                next[idx(x, y)] = c + rate * lap;
            }
        }
        field.data = next;
        }

        pub fn step(&mut self) {
            if self.paused {
                return;
            }
            // Process device events
            let events = self.devices.poll_all();
            for e in events {
                self.route_event(e);
            }
            // Compute priorities and sort
            crate::scheduler::semantic::sort_regions(&mut self.regions);
            // Compute time slices
            let slices = self.compute_time_slices();
            // Update each region
            for (i, region) in self.regions.iter().enumerate() {
                let dt = self.dt * slices.get(i).unwrap_or(&1.0);
                // Check breakpoint
                if self.breakpoints.check(&region.borrow().name) {
                    self.paused = true;
                    return;
                }
                Self::update_region(region, dt, &mut self.remote_cache);
                Self::update_priority(region);
            }
            // Prune dead regions
            for region in &self.regions {
                Self::prune_dead(region);
            }
            // Update world model
            self.world_model = build_world_model(self);
            // Apply evolution rules
            for rule in &self.evo_rules {
                for r in &self.regions {
                    evolve(&mut r.borrow_mut(), rule);
                }
            }
            // Enforce container limits
            enforce_container_limits(self);
            // Update profiles
            for r in &self.regions {
                let start = Instant::now();
                let elapsed = start.elapsed().as_secs_f32() * 1000.0;
                if let Some(p) = self.profiles.iter_mut().find(|p| p.name == r.borrow().name) {
                    p.record(self.dt, elapsed);
                } else {
                    let mut p = RegionProfile::new(&r.borrow().name);
                    p.record(self.dt, elapsed);
                    self.profiles.push(p);
                }
            }
            // Step VMs
            for vm in &self.vms {
                vm.step(self.dt);
            }
            // Update distributed scheduler
            self.distributed_scheduler.tick(self.dt as f64);
            self.distributed_scheduler.correct_drift();
            self.time += self.dt;
            self.tick += 1;
        }
        fn update_region(
            region: &Rc<RefCell<Region>>,
            dt: f32,
            remote_cache: &mut HashMap<String, MemoryField>,
        ) {
            {
                let mut r = region.borrow_mut();
                // Apply events
                apply_events(&mut r);
                apply_ui_events(&mut r);
                // Check exec capability
                if r.capabilities.exec {
                    r.behavior.apply(&mut r.memory, &r.resource, dt);
                }
                // Resource decay
                r.resource.decay(0.001 * dt);
                // Update entropy/stability
                r.entropy = crate::analysis::entropy(&r);
                r.stability = 0.5; // placeholder
                // Update resources
                crate::economy::update_resources(&mut r, dt);
                // Cognitive agent
                if let Some(agent) = &mut r.cognitive_agent {
                    think(&mut agent.cognition, &crate::kernel::build_world_model(&crate::kernel::Kernel::new(0.01)));
                    if let Some(action) = choose_action(&mut agent.cognition) {
                        execute_action(&mut r, &action);
                        agent.cognition.working.push(&format!("did {}", action));
                    }
                }
                // Agent behavior
                if let (Some(agent), Some(beh)) = (&mut r.agent, &r.agent_behavior) {
                    beh.act(agent, &mut r);
                }
                // AI behavior
                if let Some(ai) = &r.ai {
                    apply_ai(&mut r, ai.as_ref());
                }
                // Log
                r.log(crate::kernel::Kernel::new(0.01), "update", "region updated");
            }
            // Boundary mixing with parent
            {
                let mut r = region.borrow_mut();
                if let Some(parent_rc) = r.parent.upgrade() {
                    let parent = parent_rc.borrow();
                    let p_mem = &parent.memory.data;
                    let r_mem = &mut r.memory.data;
                    let b = r.boundary.permeability;
                    let inv = 1.0 - b;
                    for i in 0..r_mem.len() {
                        r_mem[i] = b * r_mem[i] + inv * p_mem[i];
                    }
                }
                // Remote mixing
                if let Some(remote_mem) = remote_cache.get(&r.name) {
                    distributed_mix(&mut r.memory, remote_mem, r.boundary.permeability);
                }
            }
            // Recurse into children
            for child in region.borrow().children.iter() {
                Self::update_region(child, dt, remote_cache);
            }
        }
        fn prune_dead(region: &Rc<RefCell<Region>>) {
            let mut r = region.borrow_mut();
            r.children.retain(|child| {
                let alive = child.borrow().resource.value > 0.0;
                if alive {
                    Self::prune_dead(child);
                }
                alive
            });
        }
        fn update_priority(region: &Rc<RefCell<Region>>) {
            let load: f32 = region.borrow().memory.data.iter().map(|v| v.abs()).sum();
            region.borrow_mut().priority = 1.0 + load * 0.01;
        }
        fn compute_time_slices(&self) -> Vec<f32> {
            let total: f32 = self.regions.iter().map(|r| r.borrow().priority).sum();
            if total == 0.0 {
                return vec![1.0; self.regions.len()];
            }
            self.regions
                .iter()
                .map(|r| r.borrow().priority / total)
                .collect()
        }

        pub struct MandalaCPU {
        pub center: Coord,
        pub radius: i32,
        pub rings: HashMap<i32, Vec<MandalaCell>>,
        pub registers: HashMap<i32, f64>,
        pub pc: HashMap<i32, usize>,
        pub memory: HashMap<Coord, f64>,
        }
        impl MandalaCPU {
        pub fn new(center: Coord, radius: i32) -> Self {
        let mut rings = HashMap::new();
        for r in 0..=radius {
        rings.insert(r, Vec::new());
        }
        Self {
        center,
        radius,
        rings,
        registers: HashMap::new(),
        pc: HashMap::new(),
        memory: HashMap::new(),
        }
        }
        pub fn tick(&mut self) {
            for (ring, cells) in self.rings.iter_mut() {
                let pc = self.pc.entry(*ring).or_insert(0);
                if *pc < cells.len() {
                    let cell = &cells[*pc];
                    match cell.opcode {
                        0 => {
                            self.registers.insert(*ring, cell.arg);
                        }
                        1 => {
                            let val = self.registers.get(ring).unwrap_or(&0.0);
                            self.memory.insert((cell.x, cell.y), val + cell.arg);
                        }
                        _ => {}
                    }
                    *pc += 1;
                }
            }
        }
        pub fn snapshot(&self) -> serde_json::Value {
            serde_json::to_value(self).unwrap()
        }
        }

            for rname in &c.regions {

                    let rb = p.region.borrow();
                    total_resource += rb.resource.value;
                    max_entropy = max_entropy.max(rb.entropy);

                    if let Some(p) = kernel.find_process(rname) {
                        p.region.borrow_mut().resource.value *= 0.9;

            }
            if max_entropy > c.entropy_cap {
                for rname in &c.regions {
                    if let Some(p) = kernel.find_process(rname) {
                        p.region.borrow_mut().entropy *= 0.95;

        pub fn run(&mut self, kernel: &mut Kernel, input: &str) -> String {
            let parts: Vec<_> = input.split_whitespace().map(|s| s.to_string()).collect();
            if parts.is_empty() {
                return "".into();
            }
            let cmd = parts[0].clone();
            let args = parts[1..].to_vec();

        pub fn interpret(kernel: &mut Kernel, input: &str) -> String {
        let parts: Vec<_> = input.split_whitespace().collect();
        if parts.is_empty() {
        return "".into();
        }
        match parts[0] {
            "spawn" => {
                if parts.len() > 1 && parts[1] == "region" {
                    kernel.spawn_region_random();
                    "region spawned".into()
                } else {
                    "unknown spawn target".into()
                }
            }
            "set" => {
                if parts.len() < 4 {
                    return "usage: set <field> <region> <value>".into();
                }
                let field = parts[1];
                let region = parts[2];
                let value = parts[3].parse::<f32>().unwrap_or(0.0);
                if let Some(p) = kernel.find_process(region) {
                    let mut r = p.region.borrow_mut();
                    match field {
                        "entropy" => r.entropy = value,
                        "resource" => r.resource.value = value,
                        _ => return format!("unknown field: {}", field),
                    }
                    "value set".into()
                } else {
                    "no such region".into()
                }
            }
            "goal" => {
                if parts.len() < 3 {
                    return "usage: goal <agent> <text>".into();
                }
                let agent = parts[1];
                let text = parts[2..].join(" ");
                kernel.add_goal(agent, &text);
                "goal added".into()
            }
            "tick" => {
                kernel.step();
                "tick".into()
            }
            "list" => {
                kernel.regions
                    .iter()
                    .map(|r| r.borrow().name.clone())
                    .collect::<Vec<_>>()
                    .join(", ")
            }
            "universes" => {
                kernel.universes
                    .keys()
                    .cloned()
                    .collect::<Vec<_>>()
                    .join(", ")
            }
            _ => "unknown command".into(),
        }
        }

        pub fn fuse(a: &Rc<RefCell<Region>>, b: &Rc<RefCell<Region>>) -> Rc<RefCell<Region>> {
        let a_b = a.borrow();
        let b_b = b.borrow();
        let mut new_mem = a_b.memory.clone();
        for i in 0..new_mem.data.len() {
            new_mem.data[i] = 0.5 * a_b.memory.data[i] + 0.5 * b_b.memory.data[i];
        }
        let new_boundary = Boundary {
            permeability: a_b.boundary.permeability.min(b_b.boundary.permeability),
        };
        let new_resource = Resource {
            value: a_b.resource.value + b_b.resource.value,
        };
        let new_caps = a_b.capabilities.clone();
        let new_region = Region::new(
            format!("{}+{}", a_b.name, b_b.name),
            new_mem,
            new_boundary,
            new_resource,
            new_caps,
            a_b.behavior.box_clone(),
        );
        new_region
        }
        pub fn fission(
        parent: &Rc<RefCell<Region>>,
        mask: Vec<bool>,
        ) -> (Rc<RefCell<Region>>, Rc<RefCell<Region>>) {
        let p = parent.borrow();
        let mut mem_a = p.memory.clone();
        let mut mem_b = p.memory.clone();
        for i in 0..p.memory.data.len() {
            if mask.get(i).copied().unwrap_or(false) {
                mem_b.data[i] = 0.0;
            } else {
                mem_a.data[i] = 0.0;
            }
        }
        let a = Region::new(
            format!("{}-A", p.name),
            mem_a,
            p.boundary.clone(),
            Resource { value: p.resource.value * 0.5 },
            p.capabilities.clone(),
            p.behavior.box_clone(),
        );
        let b = Region::new(
            format!("{}-B", p.name),
            mem_b,
            p.boundary.clone(),
            Resource { value: p.resource.value * 0.5 },
            p.capabilities.clone(),
            p.behavior.box_clone(),
        );
        (a, b)
        }

        let mut s = 0.0;

        s -= x * x.ln();

        pub fn stability(prev: &Region, curr: &Region) -> f32 {
        let mut sum = 0.0;
        for i in 0..prev.memory.data.len() {
        let d = curr.memory.data[i] - prev.memory.data[i];
        sum += d * d;
        }
        sum.sqrt()
        }
        pub fn is_converged(prev: &Region, curr: &Region, eps: f32) -> bool {
        stability(prev, curr) < eps
        }

        let e = region.entropy;
        let growth = (1.0 - (e / 100.0)).max(0.0);
        let decay = (e / 100.0).min(1.0);
        region.resource.value += growth * dt;
        region.resource.value -= decay * dt;

            region.resource.value = 0.0;
        }
        }

        pub nodes: Vec<NodeInfo>,

        }
        impl DistributedScheduler {
        pub fn new() -> Self {
        Self {
        nodes: vec![],
        drift_tolerance: 0.01,
        }
        }
        pub fn register_node(&mut self, id: &str, weight: f32) {
            self.nodes.push(NodeInfo {
                id: id.to_string(),
                weight,
                logical_time: 0.0,
            });
        }
        pub fn compute_global_dt(&self, base_dt: f64) -> f64 {
            let total_weight: f32 = self.nodes.iter().map(|n| n.weight).sum();
            base_dt * (self.nodes.len() as f64) / (total_weight as f64)
        }
        pub fn tick(&mut self, base_dt: f64) {
            let gdt = self.compute_global_dt(base_dt);
            for n in &mut self.nodes {
                n.logical_time += gdt * (n.weight as f64);
            }
        }
        pub fn correct_drift(&mut self) {
            let avg: f64 = self.nodes.iter().map(|n| n.logical_time).sum::<f64>()
                / (self.nodes.len() as f64);
            for n in &mut self.nodes {
                let diff = avg - n.logical_time;
                if diff.abs() > self.drift_tolerance {
                    n.logical_time += diff * 0.5;
                }
            }
        }
        }

            &self,

            let mem = MemoryField {
                data: self.memory.clone(),
                width: self.width,
                height: self.height,
                channels: self.channels,
            };
            let caps = Capabilities {
                read: true,
                write: true,
                exec: true,
                spawn: true,
                signal: true,
            };
            let region = Region::new(
                self.name.clone(),
                mem,
                Boundary {
                    permeability: self.boundary,
                },
                Resource { value: self.resource },
                caps,
                behavior_factory(&self.name),
            );
            {
                let mut r_mut = region.borrow_mut();
                for child_snap in &self.children {
                    let child = child_snap.to_region(behavior_factory);
                    Region::add_child(&region, child);
                }
            }
            region
        }
        }

        pub enum FsNode {
        File {
        region: Rc<RefCell<Region>>,
        },
        Directory {
        children: Vec<FsEntry>,
        },
        }
        #[derive(Clone)]
        pub struct FsEntry {
        pub name: String,
        pub node: FsNode,
        }
        pub struct FileSystem {
        pub root: FsEntry,
        }
        impl FileSystem {
        pub fn new() -> Self {
        Self {
        root: FsEntry {
        name: "/".to_string(),
        node: FsNode::Directory { children: vec![] },
        },
        }
        }
        pub fn create_file(
            dir: &mut FsEntry,
            name: &str,
            region: Rc<RefCell<Region>>,
        ) {
            if let FsNode::Directory { children } = &mut dir.node {
                children.push(FsEntry {
                    name: name.to_string(),
                    node: FsNode::File { region },
                });
            }
        }
        pub fn create_dir(dir: &mut FsEntry, name: &str) {
            if let FsNode::Directory { children } = &mut dir.node {
                children.push(FsEntry {
                    name: name.to_string(),
                    node: FsNode::Directory { children: vec![] },
                });
            }
        }
        pub fn read_file(entry: &FsEntry) -> Option<Vec<u8>> {
            if let FsNode::File { region } = &entry.node {
                let r = region.borrow();
                let mut out = Vec::new();
                for v in &r.memory.data {
                    out.extend_from_slice(&v.to_le_bytes());
                }
                return Some(out);
            }
            None
        }
        pub fn write_file(entry: &FsEntry, bytes: &[u8]) {
            if let FsNode::File { region } = &entry.node {
                let mut r = region.borrow_mut();
                let floats = bytes
                    .chunks_exact(4)
                    .map(|b| f32::from_le_bytes([b[0], b[1], b[2], b[3]]))
                    .collect::<Vec<f32>>();
                for (i, v) in floats.iter().enumerate() {
                    if i < r.memory.data.len() {
                        r.memory.data[i] = *v;
                    }
                }
            }
        }
        pub fn read_secure(entry: &FsEntry, region: &Region) -> Option<Vec<u8>> {
            if !has_cap(region, "read") {
                return None;
            }
            Self::read_file(entry)
        }
        pub fn write_secure(entry: &FsEntry, region: &Region, bytes: &[u8]) {
            if !has_cap(region, "write") {
                return;
            }
            Self::write_file(entry, bytes);
        }
        }

        pub fn execute_shell_command(kernel: &mut Kernel, cmd: ShellCommand) -> String {
        match cmd {
        ShellCommand::List => {
        kernel
        .processes
        .iter()
        .map(|p| format!("{} ({})", p.name, p.pid))
        .collect::<Vec<_>>()
        .join("\n")
        }
            ShellCommand::Kill(name) => {
                if let Some(p) = kernel.find_process(&name) {
                    kernel.kill_process(p.pid);
                    format!("killed {}", name)
                } else {
                    format!("no such process {}", name)
                }
            }
            ShellCommand::Inspect(name) => {
                if let Some(p) = kernel.find_process(&name) {
                    let r = p.region.borrow();
                    format!(
                        "process {}\npid {}\nentropy {}\nstability {}\nresource {}",
                        p.name, p.pid, r.entropy, r.stability, r.resource.value
                    )
                } else {
                    format!("no such process {}", name)
                }
            }
            ShellCommand::Reload { process, behavior } => {
                if let Some(p) = kernel.find_process(&process) {
                    p.region.borrow_mut().hot_swap_behavior(&behavior);
                    format!("reloaded {} with behavior {}", process, behavior)
                } else {
                    format!("no such process {}", process)
                }
            }
            ShellCommand::Logs { region, category } => {
                let r = region.as_deref();
                let c = category.as_deref();
                let events = query_logs(&kernel.log_buffer, r, c);
                events
                    .iter()
                    .map(|e| format!("[{}] {}: {}", e.timestamp, e.region, e.message))
                    .collect::<Vec<_>>()
                    .join("\n")
            }
            ShellCommand::Audit(name) => {
                let events = &kernel.capability_audit.events;
                let filtered: Vec<_> = events
                    .iter()
                    .filter(|e| e.region == name)
                    .collect();
                filtered
                    .iter()
                    .map(|e| format!("[{}] {} {} {}", e.timestamp, e.region, e.capability, e.action))
                    .collect::<Vec<_>>()
                    .join("\n")
            }
            ShellCommand::BpSet(name) => {
                kernel.breakpoints.set(&name);
                format!("breakpoint set on {}", name)
            }
            ShellCommand::BpClear(name) => {
                kernel.breakpoints.clear(&name);
                format!("breakpoint cleared on {}", name)
            }
            ShellCommand::Pause => {
                kernel.paused = true;
                "paused".to_string()
            }
            ShellCommand::Resume => {
                kernel.paused = false;
                "resumed".to_string()
            }
            ShellCommand::Step => {
                kernel.step_with_dt(kernel.dt);
                "stepped".to_string()
            }
            ShellCommand::Heal(name) => {
                if let Some(p) = kernel.find_process(&name) {
                    heal_region(&mut p.region.borrow_mut());
                    format!("healed {}", name)
                } else {
                    format!("no such process {}", name)
                }
            }
            ShellCommand::SymGet { region, key } => {
                if let Some(p) = kernel.find_process(&region) {
                    p.region.borrow().symbolic.get(&key).unwrap_or("none".to_string())
                } else {
                    "no such region".to_string()
                }
            }
            ShellCommand::SymSet { region, key, value } => {
                if let Some(p) = kernel.find_process(&region) {
                    p.region.borrow_mut().symbolic.set(&key, &value);
                    "ok".to_string()
                } else {
                    "no such region".to_string()
                }
            }
            ShellCommand::AiAttach { region, kind } => {
                if let Some(p) = kernel.find_process(&region) {
                    if kind == "adaptive" {
                        p.region.borrow_mut().ai = Some(Box::new(AdaptiveBehavior));
                        return "ai attached".to_string();
                    }
                }
                "failed".to_string()
            }
            ShellCommand::AiDetach(region) => {
                if let Some(p) = kernel.find_process(&region) {
                    p.region.borrow_mut().ai = None;
                    return "ai detached".to_string();
                }
                "failed".to_string()

            ShellCommand::CogCreate { region, name } => {

                    p.region.borrow_mut().cognitive_agent = Some(CognitiveAgent {

                            "agent {}: goals={:?}, subgoals={:?}, working={:?}",

                    p.region.borrow_mut().agent = Some(Agent {
                        name,
                        goals: vec![],
                        memory: vec![],

                    p.region.borrow_mut().agent_behavior = Some(Box::new(GoalBehavior));

                    "no such region".to_string()

                        a.goals.push(goal);

                    if let Some(a) = &p.region.borrow().agent {

class PersistentStore:
    def __init__(self, db_path: str = "validator.db"):
        self.db_path = db_path
        self._init_db()

    def _init_db(self):
        with sqlite3.connect(self.db_path) as conn:
            conn.execute("""
                CREATE TABLE IF NOT EXISTS nodes (
                    id TEXT PRIMARY KEY,
                    parent_id TEXT,
                    trait TEXT,
                    color TEXT,
                    leela TEXT,
                    music TEXT,
                    signal TEXT,
                    fractal_echo TEXT,
                    timestamp TEXT,
                    watermark TEXT,
                    glyphs TEXT,
                    metadata TEXT,
                    validated_at TEXT
                )
            """)

    def save_node(self, node: NodeState):
        with sqlite3.connect(self.db_path) as conn:
            conn.execute("""
                INSERT OR REPLACE INTO nodes (id, parent_id, trait, color, leela, music, signal,
                    fractal_echo, timestamp, watermark, glyphs, metadata, validated_at)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
            """, (
                node.id, node.parent_id, node.trait, node.color, node.leela, node.music,
                node.signal, node.fractal_echo, node.timestamp,
                json.dumps(node.watermark), json.dumps(node.glyphs),
                json.dumps(node.metadata), datetime.utcnow().isoformat()
            ))

    def load_node(self, node_id: str) -> Optional[NodeState]:
        with sqlite3.connect(self.db_path) as conn:
            cur = conn.execute("SELECT * FROM nodes WHERE id=?", (node_id,))
            row = cur.fetchone()
            if row:
                return NodeState(
                    id=row[0], parent_id=row[1], trait=row[2], color=row[3],
                    leela=row[4], music=row[5], signal=row[6], fractal_echo=row[7],
                    timestamp=row[8], watermark=json.loads(row[9]),
                    glyphs=json.loads(row[10]), metadata=json.loads(row[11])
                )
        return None

    def load_all_nodes(self) -> List[NodeState]:
        nodes = []
        with sqlite3.connect(self.db_path) as conn:
            cur = conn.execute("SELECT * FROM nodes")
            for row in cur.fetchall():
                nodes.append(NodeState(
                    id=row[0], parent_id=row[1], trait=row[2], color=row[3],
                    leela=row[4], music=row[5], signal=row[6], fractal_echo=row[7],
                    timestamp=row[8], watermark=json.loads(row[9]),
                    glyphs=json.loads(row[10]), metadata=json.loads(row[11])
                ))
        return nodes

class UrbitBridge:
    def __init__(self, ship_url: str, code: str, use_websocket: bool = False):
        self.base_url = ship_url.rstrip('/')
        self.session = requests.Session()
        self.code = code
        self.use_websocket = use_websocket
        self.ws = None
        self._authenticate()
        self.subscription_id = None

    def _authenticate(self):
        # Standard login
        resp = self.session.post(
            f"{self.base_url}/~/login",
            data={"password": self.code},
            headers={"Content-Type": "application/x-www-form-urlencoded"}
        )
        if resp.status_code != 200:
            raise RuntimeError("Urbit auth failed")

    def send_event(self, app: str, mark: str, data: Dict):
        # Send via HTTP POST (or WebSocket if enabled)
        if self.use_websocket and self.ws:
            self.ws.send(json.dumps({"type": "event", "app": app, "mark": mark, "data": data}))
        else:
            resp = self.session.post(
                f"{self.base_url}/~/channel",
                json={"source": {"app": app}, "event": data, "mark": mark},
                headers={"Content-Type": "application/json"}
            )
            if resp.status_code not in (200, 201):
                raise RuntimeError(f"Failed to send event: {resp.text}")

    def connect_websocket(self, on_message):
        """Connect to Urbit's WebSocket endpoint."""
        ws_url = self.base_url.replace("http", "ws") + "/~/ws"
        self.ws = websocket.WebSocketApp(
            ws_url,
            on_open=lambda ws: self._on_open(ws),
            on_message=lambda ws, msg: on_message(json.loads(msg)),
            on_error=lambda ws, err: print(f"WebSocket error: {err}"),
            on_close=lambda ws, code, msg: print("WebSocket closed")
        )
        threading.Thread(target=self.ws.run_forever, daemon=True).start()

    def _on_open(self, ws):
        # Send authentication and subscribe
        ws.send(json.dumps({"type": "auth", "code": self.code}))
        # Subscribe to validator app
        ws.send(json.dumps({"type": "subscribe", "app": "validator", "path": "/validation"}))

import random
from collections import defaultdict

class FractalSynthesizer:
    def __init__(self, corpus: List[str]):
        self.chain = defaultdict(list)
        for phrase in corpus:
            tokens = phrase.split(" → ")
            for i in range(len(tokens)-1):
                self.chain[tokens[i]].append(tokens[i+1])
        self.start_tokens = [t for t in self.chain if t]

    def synthesize(self, seed: str) -> str:
        # Start with seed or random start
        current = seed if seed in self.chain else random.choice(self.start_tokens)
        result = [current]
        for _ in range(4):  # generate 5 steps
            if current in self.chain:
                next_token = random.choice(self.chain[current])
                result.append(next_token)
                current = next_token
            else:
                break
        return " → ".join(result)
