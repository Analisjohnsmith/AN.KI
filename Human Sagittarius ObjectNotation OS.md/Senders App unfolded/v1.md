fn parse_add_sub(tokens: &[String], pos: &mut usize, state: &ContainerState) -> Result<f64, String> {
    let mut val = parse_mul_div(tokens, pos, state)?;

    while *pos < tokens.len() {

            *pos += 1;

pub fn validate_instance(ts: &TypeSystem, inst: &Instance) -> Result<(), String> {
    let ty = ts.get(&inst.ty).ok_or("Unknown type")?;

    for field in &ty.fields {
        let val = inst.fields.get(&field.name)
            .ok_or(format!("Missing field {}", field.name))?;

        match (&field.ty, val) {
            (FieldType::F64, Value::F64(_)) => {}
            (FieldType::I64, Value::I64(_)) => {}
            (FieldType::Bool, Value::Bool(_)) => {}
            _ => return Err(format!("Type mismatch on field {}", field.name)),
        }
    }

    Ok(())
}

    pub fn tick(&mut self, now_ms: u64) {
        self.gates.update(&self.state);
        self.shrines.update(&self.state, &self.gates);

        self.scheduler.run(self, now_ms);

        if let Some(packet) = self.sync.receive() {
            self.sync.apply(self, packet);
        }

        self.ledger.record("tick", &self.state);
    }

            timestamp: current_time_ms(),
            state: self.state.clone(),
            types: self.types.all(),
            ops: self.ops.all(),

            schedules: self.scheduler.all(),
            gates: self.gates.all(),

impl Kernel {
    pub fn restore(&mut self, snap: KernelSnapshot) {
        self.state = snap.state;
        self.types.load(snap.types);
        self.ops.load(snap.ops);
        self.wasm.load(snap.modules);
        self.scheduler.load(snap.schedules);
        self.gates.load(snap.gates);
        self.shrines.load(snap.shrines);

        self.virtual_kernels.clear();
        for vk_snap in snap.virtual_kernels {
            self.virtual_kernels.insert(vk_snap.id.clone(), VirtualKernel::restore(vk_snap));
        }
    }
}

pub fn decrypt(key: &KernelKey, blob: &[u8]) -> Vec<u8> {
    let cipher = Aes256Gcm::new(&key.key);
    let (nonce_bytes, enc_bytes) = blob.split_at(12);
    let nonce = Nonce::from_slice(nonce_bytes);
    cipher.decrypt(nonce, enc_bytes).unwrap()
}

impl PolicyEngine {
    pub fn evaluate(&self, kernel: &Kernel, ctx: &Value) -> Option<&PolicyAction> {
        'outer: for p in &self.policies {
            for cond in &p.conditions {
                match cond {
                    PolicyCondition::MetricAbove { metric, threshold } => {
                        if kernel.metrics[metric].value <= *threshold {
                            continue 'outer;
                        }
                    }
                    PolicyCondition::MetricBelow { metric, threshold } => {
                        if kernel.metrics[metric].value >= *threshold {
                            continue 'outer;
                        }
                    }
                    PolicyCondition::SpaceMatch { space } => {
                        if ctx.get("space").and_then(|v| v.as_str()) != Some(space.as_str()) {
                            continue 'outer;
                        }
                    }
                    PolicyCondition::TagMatch { key, value } => {
                        let tags = ctx.get("tags").and_then(|v| v.as_array()).unwrap_or(&vec![]);
                        let mut ok = false;
                        for t in tags {
                            if t.get("key") == Some(&Value::String(key.clone()))
                                && t.get("value") == Some(&Value::String(value.clone()))
                            {
                                ok = true;
                                break;
                            }
                        }
                        if !ok {
                            continue 'outer;
                        }
                    }
                }
            }
            return Some(&p.action);
        }
        None
    }
}

use std::io::{self, Write};
use universe_kernel::Kernel;
use parking_lot::Mutex;

pub fn start_shell(kernel: Arc<Mutex<Kernel>>) {
    let stdin = io::stdin();
    let mut stdout = io::stdout();

    loop {
        write!(stdout, "kernel> ").unwrap();
        stdout.flush().unwrap();

        let mut line = String::new();
        if stdin.read_line(&mut line).is_err() {
            break;
        }

        let cmd = line.trim();
        if cmd == "exit" {
            break;
        } else if cmd.starts_with("run ") {
            let op = &cmd[4..];
            kernel.lock().run_op(op);
        } else if cmd == "metrics" {
            let k = kernel.lock();
            for (name, m) in &k.metrics {
                println!("{} = {}", name, m.value);
            }
        } else if cmd == "graph" {
            let k = kernel.lock();
            for e in &k.graph.edges {
                println!("{} --({})--> {}", e.from, e.kind, e.to);
            }
        }
    }
}

use std::collections::HashMap;

#[derive(Debug, Clone, Copy, Eq, PartialEq, Hash)]
pub enum Domain {
    Windows,
    MacOS,
    Android,
    Linux,
    Wasm,
    Ai,
}

#[derive(Debug, Clone)]
pub enum DomainState {
    Starting,
    Running,
    Stopped,
    Crashed,
}

#[derive(Debug, Clone)]
pub struct DomainInstance {
    pub domain: Domain,
    pub pid: u32,
    pub runtime: String,
    pub state: DomainState,
}

pub struct RuntimeRegistry {
    pub windows: String,
    pub macos: String,
    pub android: String,
    pub linux: String,
    pub wasm: String,
    pub ai: String,
}

pub struct DomainManager {
    pub active: HashMap<Domain, DomainInstance>,
    pub runtimes: RuntimeRegistry,
}

impl DomainManager {
    pub fn new(runtimes: RuntimeRegistry) -> Self {
        Self {
            active: HashMap::new(),
            runtimes,
        }
    }

    pub fn start_domain(&mut self, domain: Domain) -> std::io::Result<()> {
        let runtime = match domain {
            Domain::Windows => &self.runtimes.windows,
            Domain::MacOS => &self.runtimes.macos,
            Domain::Android => &self.runtimes.android,
            Domain::Linux => &self.runtimes.linux,
            Domain::Wasm => &self.runtimes.wasm,
            Domain::Ai => &self.runtimes.ai,
        };

        let child = std::process::Command::new(runtime)
            .spawn()?;

        let instance = DomainInstance {
            domain,
            pid: child.id(),
            runtime: runtime.clone(),
            state: DomainState::Running,
        };

        self.active.insert(domain, instance);
        Ok(())
    }

    pub fn stop_domain(&mut self, domain: Domain) {
        if let Some(inst) = self.active.get(&domain) {
            let _ = nix::sys::signal::kill(
                nix::unistd::Pid::from_raw(inst.pid as i32),
                nix::sys::signal::Signal::SIGTERM,
            );
        }
        self.active.remove(&domain);
    }
}

pub struct HealthManager {
    domains: DomainManager,
}

impl HealthManager {
    pub fn tick(&mut self) {
        for (domain, instance) in self.domains.active.iter_mut() {
            if !process_alive(instance.pid) {
                instance.state = DomainState::Crashed;
                self.domains.restart_domain(*domain);
            }

            let ctx = self.domains.get_context(*domain);

            if ctx.cpu > 90 {
                self.domains.throttle_domain(*domain);
            }

            if ctx.iowait > 20 {
                self.domains.optimize_io(*domain);
            }
        }
    }
}

use sysinfo::{System, SystemExt, ProcessExt};

pub struct Metrics {
    sys: System,
}

impl Metrics {
    pub fn new() -> Self {
        let mut sys = System::new_all();
        sys.refresh_all();
        Self { sys }
    }

    pub fn domain_metrics(&mut self, pid: u32) -> DomainMetrics {
        self.sys.refresh_process(pid);

        if let Some(proc) = self.sys.process(pid as i32) {
            DomainMetrics {
                cpu: proc.cpu_usage(),
                ram: proc.memory(),
                iowait: self.sys.global_processor_info().cpu_usage(), // simplified
            }
        } else {
            DomainMetrics {
                cpu: 0.0,
                ram: 0,
                iowait: 0.0,
            }
        }
    }

    pub fn kernel_metrics(&mut self) -> KernelMetrics {
        self.sys.refresh_system();
        KernelMetrics {
            load: self.sys.load_average(),
            total_memory: self.sys.total_memory(),
            used_memory: self.sys.used_memory(),
        }
    }
}

use std::os::unix::net::{UnixListener, UnixStream};
use std::io::{Read, Write};

pub struct PersonaIO {
    in_listener: UnixListener,
    out_listener: UnixListener,
    kernel_listener: UnixListener,
}

impl PersonaIO {
    pub fn new() -> std::io::Result<Self> {
        Ok(Self {
            in_listener: UnixListener::bind("/run/hazeru/persona.in.sock")?,
            out_listener: UnixListener::bind("/run/hazeru/persona.out.sock")?,
            kernel_listener: UnixListener::bind("/run/hazeru/persona.kernel.sock")?,
        })
    }

    pub fn read_input(&self) -> std::io::Result<String> {
        let (mut stream, _) = self.in_listener.accept()?;
        let mut buf = String::new();
        stream.read_to_string(&mut buf)?;
        Ok(buf)
    }

    pub fn send_output(&self, msg: &str) -> std::io::Result<()> {
        let (mut stream, _) = self.out_listener.accept()?;
        stream.write_all(msg.as_bytes())?;
        Ok(())
    }

    pub fn send_kernel(&self, msg: &str) -> std::io::Result<()> {
        let (mut stream, _) = self.kernel_listener.accept()?;
        stream.write_all(msg.as_bytes())?;
        Ok(())
    }
}

fn main() -> std::io::Result<()> {
    // Boot orchestration
    let mut memory = Persistence::load_memory();
    let mut graph = Persistence::load_graph();
    let mut metrics = Metrics::new();
    let runtimes = RuntimeRegistry::load_default();
    let mut domains = DomainManager::new(runtimes);
    let mut persona = Persona::new(memory.clone(), graph.clone());
    let io = PersonaIO::new()?;
    let mut health = HealthManager::new(&domains);

    loop {
        // 1. Read persona input (from shell)
        let msg = io.read_input()?;

        // 2. Intent + target resolution
        let intent = classify(&msg);
        let target = extract_target(&msg);
        let resolution = resolve(intent, &target);

        // 3. Ensure domain is running
        domains.start_domain(resolution.domain)?;

        // 4. Collect metrics
        let ctx = domains.get_context(resolution.domain);
        let kernel_stats = metrics.kernel_metrics();

        // 5. Persona reasoning
        let response = persona.reason(
            intent,
            resolution.domain,
            ctx,
            &memory,
            &msg,
        );

        // 6. Update memory
        memory.update(intent, resolution.domain, ctx);

        // 7. Persist state

        Persistence::save_graph(&graph)?;

        // 9. Send output to shell

Domain detection:
rust
pub fn detect_domain(file: &str) -> Domain {
    if file.ends_with(".exe") { Domain::Windows }
    else if file.ends_with(".app") { Domain::MacOS }
    else if file.ends_with(".apk") { Domain::Android }
    else if file.ends_with(".wasm") { Domain::Wasm }
    else { Domain::Linux }
}
