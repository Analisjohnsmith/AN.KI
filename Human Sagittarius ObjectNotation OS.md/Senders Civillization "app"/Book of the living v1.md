class WowKernel:
    """Boots the archive, binds opcodes, decodes the bitstream, checkpoints."""

    def __init__(self):
        self.metrics = {
            "frequency": 1420.0,
            "duration": 72.0,
            "bitstream_load": 0.0,
            "bitstream_remaining": 0.0,
        }
        self.decoded = []          # (opcode, symbol, meaning, lab_op)
        self.recognized = set()    # set of opcodes seen
        self.booted = False

    # -- boot sequence ----------------------------------------------------
    def boot(self):
        for stage, desc in enumerate(
            ["seed_initialization", "lexicon_activation", "planet_encoding",
             "protocol_handshake", "archive_lock", "recursion_engine",
             "runtime_execution"], 1):
            print(f"  [boot] stage_{stage}: {desc}")
        self.booted = True

    # -- decoder ----------------------------------------------------------
    @staticmethod
    def _lookup(code):
        """Return (symbol, meaning, lab_op) for an opcode or None."""
        entry = OPCODES.get(code)
        if not entry:
            return None
        symbol, meaning = entry
        return symbol, meaning, LAB_OPS.get(code, "unknown_op")

    def decode_canonical(self, line):
        """Decode a space-separated line (authoritative, lossless)."""
        for tok in line.split():
            hit = self._lookup(tok)
            if hit:
                self.decoded.append((tok,) + hit)
                self.recognized.add(tok)
            else:
                self.decoded.append((tok, "RAW", "unmapped", "unknown_op"))

    def decode_flattened(self, line):
        """Greedy longest-match over a separator-less bit run."""
        i = 0
        n = len(line)
        while i < n:
            matched = None
            for length in range(min(8, n - i), 0, -1):
                chunk = line[i:i + length]
                if chunk in OPCODES:
                    hit = self._lookup(chunk)
                    matched = (chunk,) + hit
                    break
            if matched:
                self.decoded.append(matched)
                self.recognized.add(matched[0])
                i += len(matched[0])
            else:
                self.decoded.append((line[i], "RAW", "unmapped", "unknown_op"))
                i += 1

    def feed(self, lines, flattened=False):
        func = self.decode_flattened if flattened else self.decode_canonical
        for line in lines:
            func(line)
        self.metrics["bitstream_remaining"] = float(len(lines))
        self.metrics["bitstream_load"] = float(len(lines))
        print(f"  [decode] fed {len(lines)} line(s)")

    # -- schedule loop ----------------------------------------------------
    def schedule_loop(self, interval_ms=100):
        passes = 0
        while self.metrics["bitstream_remaining"] > 0 and passes < 500:
            self.metrics["bitstream_remaining"] -= 1
            self.metrics["bitstream_load"] = self.metrics["bitstream_remaining"]
            passes += 1
        print(f"  [loop]   {passes} pass(es) @ {interval_ms}ms interval")
        print(f"  [gate]   bitstream_remaining={int(self.metrics['bitstream_remaining'])}"
              f" -> decode-complete={self.metrics['bitstream_remaining'] <= 0}")

    # -- output -----------------------------------------------------------
    def report(self):
        print("\n=== DECODED ARCHIVE (usable form) ===")
        seen = set()
        for code, symbol, meaning, lab_op in self.decoded:
            key = (code, symbol)
            if key in seen:
                continue
            seen.add(key)
            print(f"  {code:<6} -> {symbol:<22} : {meaning:<12} -> {lab_op}")
        print(f"\n  {len(self.recognized)} unique opcodes recognized")
        print(f"  {len(self.decoded)} total opcode tokens decoded")

    def checkpoint(self, path="wowkernel_checkpoint.json"):

            "schema": "wowkernel/1",
            "timestamp": datetime.now(timezone.utc).isoformat(),
