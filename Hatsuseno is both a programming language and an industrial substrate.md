    def wrap_text(text, font, max_w):
        words, lines, cur = text.split(), [], ""
        for wd in words:
            trial = (cur + " " + wd).strip()
            if font.getlength(trial) <= max_w or not cur:
                cur = trial
            else:
                lines.append(cur)
                cur = wd
        if cur:
            lines.append(cur)
        return lines

    def text_block(d, x0, y0, x1, text, font, fill, lh=None, anchor="la", align="left"):

        lh = lh or int(font.size * 1.25)
        for ln in lines:
            if align == "center":
                d.text(((x0 + x1) / 2, y0), ln, font=font, fill=fill, anchor=anchor)
            else:
                d.text((x0, y0), ln, font=font, fill=fill, anchor=anchor)
            y0 += lh
        return y0

    def make_halftone(w, h, spacing=26, intensity=0.05, seed=0):
        r = np.random.default_rng(seed)
        arr = np.zeros((h, w), np.float32)
        for i in range(0, h + spacing, spacing):

        return arr
